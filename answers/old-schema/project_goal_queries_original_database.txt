-- Question Set 1 - Easy

-- 1. Who is the most senior employee based on job title?

-- NOTE: There are multiple ways to find the most senior employee.
-- We can use columns "title", "report_to", "levels", "hire_date".

-- Solution 1: Using "title".
SELECT
    *
FROM "employee"
WHERE "title" = 'Senior General Manager';

-- Solution 2: Using "reports_to".
SELECT
    *
FROM "employee"
WHERE "reports_to" IS NULL;

-- Solution 3: Using "levels".
SELECT
	*
FROM "employee"
ORDER BY "levels" DESC
LIMIT 1;

-- Solution 4: Using "hire_date".
SELECT
	*
FROM "employee"
ORDER BY "hire_date" ASC
LIMIT 1;

-- 2. Which countries have the most invoices?

-- Solution: Using GROUP BY.
SELECT
	"billing_country",
	COUNT(*) AS "count_of_invoices_per_country"
FROM "invoice"
GROUP BY "billing_country"
ORDER BY COUNT(*) DESC;

-- 3. What are the top 3 values of total invoice?

-- NOTE: The "total" column has the 'double precision' datatype,
-- which can't be used alongside the ROUND() function.
-- I needed to CAST the values into a decimal number using either DECIMAL() or NUMERIC().

-- Solution: Using LIMIT.
SELECT
	ROUND(CAST("total" AS DECIMAL), 2) AS "rounded_total_of_invoice"
FROM "invoice"
ORDER BY "total" DESC
LIMIT 3;

-- 4. What city has the best customers?
-- We would like to organize a Music Festival in the city that produces
-- the higher earnings. Write a query that returns the city with the
-- highest sum of invoice totals. The query should return the city's name
-- and its invoice totals.

-- Solution 1: Using GROUP BY and LIMIT.
SELECT
	"billing_city",
	SUM(CAST("total" AS DECIMAL)) AS "total_city_earnings"
FROM "invoice"
GROUP BY "billing_city"
ORDER BY "total_city_earnings" DESC
LIMIT 1;

-- Solution 2: Using an uncorrelated subquery and CTE.
SELECT
	"billing_city",
	SUM(CAST("total" AS DECIMAL)) AS "total_city_earnings"
FROM "invoice"
GROUP BY "billing_city"
HAVING SUM(CAST("total" AS DECIMAL)) = (WITH "cte_city_max_earnings" AS(
	                                       SELECT
	                                           "billing_city",
		                                        SUM(CAST("total" AS DECIMAL)) AS "total_city_earnings"
	                                        FROM "invoice"
	                                        GROUP BY "billing_city"
                                        )
                                        SELECT
                                            MAX("total_city_earnings")
                                        FROM "cte_city_max_earnings");

-- 5. Who is the best customer?
-- The customer who has spent the most money will be declared the best.
-- Write a query that returns the name of that customer and the sum amount
-- he or she has spent.

-- Solution 1: Using GROUP BY and LIMIT.
SELECT
    "customer"."customer_id",
    TRIM("customer"."first_name") AS "customer_first_name",
	TRIM("customer"."last_name") AS "customer_last_name",
	SUM(CAST("invoice"."total" AS DECIMAL)) AS "customer_spendings"
FROM "invoice"
LEFT OUTER JOIN "customer"
	ON "invoice"."customer_id" = "customer"."customer_id"
GROUP BY "customer"."customer_id", "customer"."first_name", "customer"."last_name"
ORDER BY "customer_spendings" DESC
LIMIT 1;

-- Solution 2: Using uncorrelated subqueries.
SELECT
    "customer"."customer_id",
    TRIM("customer"."first_name") AS "customer_first_name",
	TRIM("customer"."last_name") AS "customer_last_name",
	SUM(CAST("invoice"."total" AS DECIMAL)) AS "customer_spendings"
FROM "invoice"
LEFT OUTER JOIN "customer"
	ON "invoice"."customer_id" = "customer"."customer_id"
GROUP BY "customer"."customer_id", "customer"."first_name", "customer"."last_name"
HAVING SUM(CAST("total" AS DECIMAL)) = (SELECT
                                            MAX("customer_spendings")
										FROM(SELECT
                                                SUM(CAST("invoice"."total" AS DECIMAL)) AS "customer_spendings"
										     FROM "invoice"
										     LEFT OUTER JOIN "customer"
	                                             ON "invoice"."customer_id" = "customer"."customer_id"
										     GROUP BY "customer"."customer_id"));

-- Question Set 2 - Moderate

-- 1. Write a query that returns the email, first name, last name and genre
-- of all Rock music listeners. Return the list ordered alphabetically by email
-- starting with 'A'.

-- Solution: Using GROUP BY to remove duplicate values, instead of DISTINCT.
SELECT
	"customer"."email",
    "customer"."first_name",
	"customer"."last_name",
	"genre"."name" AS "music_genre"
FROM "customer"
LEFT OUTER JOIN "invoice"
	ON "customer"."customer_id" = "invoice"."customer_id"
LEFT OUTER JOIN "invoice_line"
	ON "invoice"."invoice_id" = "invoice_line"."invoice_id"
LEFT OUTER JOIN "track"
	ON "invoice_line"."track_id" = "track"."track_id"
LEFT OUTER JOIN "genre"
	ON "track"."genre_id" = "genre"."genre_id"
WHERE "genre"."name" = 'Rock'
GROUP BY "customer"."email", "customer"."first_name", "customer"."last_name", "genre"."name"
ORDER BY "customer"."email" ASC;

-- 2. Let's invite the artist who has written the most rock tracks,
-- appearing on our database. Write a query that returns the Artist's name
-- and total track count of the top 10 rock bands.

-- Solution: Using GROUP BY and LIMIT.
SELECT
	"artist"."artist_id",
	"artist"."name",
	COUNT("track"."track_id") AS "count_of_written_tracks"
FROM "artist"
LEFT OUTER JOIN "album"
	ON "artist"."artist_id" = "album"."artist_id"
LEFT OUTER JOIN "track"
	ON "album"."album_id" = "track"."album_id"
LEFT OUTER JOIN "genre"
	ON "track"."genre_id" = "genre"."genre_id"
WHERE "genre"."name" = 'Rock'
GROUP BY "artist"."artist_id"
ORDER BY "count_of_written_tracks" DESC
LIMIT 10;

-- 3. Return all the track names that have a song length longer that the average
-- song length. Return the Name and Milliseconds for each track. Order by the
-- song length with the longest songs listed first.

-- Solution: Using uncorrelated subquery.
SELECT
	"track_id",
	"name" AS "track_name", 
	"milliseconds" AS "track_duration_in_milliseconds"
FROM "track"
WHERE "milliseconds" > (SELECT
							AVG("milliseconds")
						FROM "track")
ORDER BY "milliseconds" DESC;

-- Question Set 3 - Advanced

-- 1. Find how much money was spend by each customer on the top selling artist.
-- Write a query to return customer name, artist name and total spent.

-- Solution 1: Using uncorrelated subquery and LIMIT.
SELECT
	"customer"."customer_id",
	"customer"."first_name" || ' ' || "customer"."last_name" AS "customer_name",
	"artist"."name" AS "top_selling_artist",
	SUM(CAST("invoice_line"."unit_price" * "invoice_line"."quantity" AS DECIMAL)) AS "total_amount_spent"
FROM "customer"
LEFT OUTER JOIN "invoice"
	ON "customer"."customer_id" = "invoice"."customer_id"
LEFT OUTER JOIN "invoice_line"
	ON "invoice"."invoice_id" = "invoice_line"."invoice_id"
LEFT OUTER JOIN "track"
	ON "invoice_line"."track_id" = "track"."track_id"
LEFT OUTER JOIN "album"
	ON "track"."album_id" = "album"."album_id"
LEFT OUTER JOIN "artist"
	ON "album"."artist_id" = "artist"."artist_id"
WHERE "artist"."artist_id" = (SELECT
								  "artist_id"
							  FROM(SELECT
							  		   "artist"."artist_id",
							           SUM(CAST("invoice_line"."unit_price" * "invoice_line"."quantity" AS DECIMAL)) AS "total_artist_earnings"
								   FROM "invoice_line"
								   INNER JOIN "track"
								       ON "invoice_line"."track_id" = "track"."track_id"
								   INNER JOIN "album"
									   ON "track"."album_id" = "album"."album_id"
								   INNER JOIN "artist"
									   ON "album"."artist_id" = "artist"."artist_id"
								   GROUP BY "artist"."artist_id"
								   ORDER BY "total_artist_earnings" DESC
								   LIMIT 1)
							  )
GROUP BY "customer"."customer_id", "top_selling_artist"
ORDER BY "total_amount_spent" DESC;

-- Solution 2: Using uncorrelated subqueries without LIMIT.

							  HAVING SUM(CAST("invoice_line"."unit_price" * "invoice_line"."quantity" AS DECIMAL)) = (SELECT
								                                                                                          MAX("total_artist_earnings")
								   																					  FROM(SELECT
							  	   	                                                                                           SUM(CAST("invoice_line"."unit_price" * "invoice_line"."quantity" AS DECIMAL)) AS "total_artist_earnings"
								                                                                                           FROM "invoice_line"

								    																				 )

-- Solution 3: Using CTE and LIMIT.
WITH "top_selling_artist_cte" AS(
	SELECT
		"artist"."artist_id",
		"artist"."name" AS "artist_name",
		SUM(CAST("invoice_line"."unit_price" * "invoice_line"."quantity" AS DECIMAL)) AS "total_artist_earnings"
	FROM "invoice_line"
	INNER JOIN "track"
		ON "invoice_line"."track_id" = "track"."track_id"
	INNER JOIN "album"
		ON "track"."album_id" = "album"."album_id"
	INNER JOIN "artist"
		ON "album"."artist_id" = "artist"."artist_id"
	GROUP BY "artist"."artist_id"
	ORDER BY SUM("invoice_line"."unit_price" * "invoice_line"."quantity") DESC
	LIMIT 1
)
SELECT
	"customer"."customer_id",
	TRIM("customer"."first_name") AS "customer_first_name",
	TRIM("customer"."last_name") AS "customer_last_name",
	"top_selling_artist_cte"."artist_name" AS "top_selling_artist",
	SUM(CAST("invoice_line"."unit_price" * "invoice_line"."quantity" AS DECIMAL)) AS "total_amount_spent"
FROM "invoice"
INNER JOIN "customer"
	ON "invoice"."customer_id" = "customer"."customer_id"
INNER JOIN "invoice_line"
	ON "invoice_line"."invoice_id" = "invoice"."invoice_id"
INNER JOIN "track"
	ON "invoice_line"."track_id" = "track"."track_id"
INNER JOIN "album"
	ON "track"."album_id" = "album"."album_id"
INNER JOIN "top_selling_artist_cte"
	ON "album"."artist_id" = "top_selling_artist_cte"."artist_id"
GROUP BY "customer"."customer_id", "customer"."first_name", "customer"."last_name", "top_selling_artist"
ORDER BY "total_amount_spent" DESC;

-- 2. We want to find out the most popular music genre for each country. 
-- We determine the most popular genre as the genre with the highest amount
-- of purchases. Write a query that returns each country along with the top Genre.
-- For countries where the maximum number of puchases is shared, return all genres.

-- Solution: Using multiple CTEs and WINDOW FUNCTIONS.
WITH "earnings_cte" AS(
    SELECT
        "invoice"."billing_country" AS "country",
		"genre"."genre_id",
	    "genre"."name" AS "genre_name",
         SUM("invoice_line"."unit_price" * "invoice_line"."quantity") AS "revenue_per_genre"
    FROM "invoice"
    LEFT OUTER JOIN "invoice_line"
        ON "invoice"."invoice_id" = "invoice_line"."invoice_id"
    LEFT OUTER JOIN "track"
        ON "invoice_line"."track_id" = "track"."track_id"
    LEFT OUTER JOIN "genre"
        ON "track"."genre_id" = "genre"."genre_id"
    GROUP BY "invoice"."billing_country", "genre"."genre_id", "genre"."name"
),
"ranked_country_earnings_cte" AS(
SELECT
    "earnings_cte"."country",
	"earnings_cte"."genre_name",
	"earnings_cte"."revenue_per_genre",
	RANK() OVER(PARTITION BY "earnings_cte"."country" ORDER BY "earnings_cte"."revenue_per_genre" DESC) AS "ranking"
FROM "earnings_cte"
)
SELECT
    "ranked_country_earnings_cte"."country",
	"ranked_country_earnings_cte"."genre_name"
FROM "ranked_country_earnings_cte"
WHERE "ranked_country_earnings_cte"."ranking" = 1;

-- 3. Write a query that determines the customer that has spent the most on music
-- for each country. Write a query that returns the country along with the top
-- customer and how much they spent. For countries where the top amount spent is
-- shared, provide all customers who spent this amount. 

-- Solution: Using CTE and WINDOW FUNCTIONS.
WITH "customer_spendings_per_country" AS(
	SELECT
    	"invoice"."billing_country" AS "country",
		"customer"."customer_id",
		TRIM("customer"."first_name") AS "first_name",
		TRIM("customer"."last_name") AS "last_name",
		SUM(CAST("invoice"."total" AS DECIMAL)) AS "total_spendings",
		RANK() OVER(PARTITION BY "invoice"."billing_country" ORDER BY SUM("invoice"."total") DESC) AS "ranking"
	FROM "customer"
	LEFT OUTER JOIN "invoice"
    	ON "customer"."customer_id" = "invoice"."customer_id"
	GROUP BY "invoice"."billing_country", "customer"."customer_id", "customer"."first_name", "customer"."last_name"
)
SELECT
	"cspc"."country",
	"cspc"."first_name" AS "customer_first_name",
	"cspc"."last_name" AS "customer_last_name",
	"cspc"."total_spendings"
FROM "customer_spendings_per_country" AS "cspc"
WHERE "ranking" = 1
ORDER BY "total_spendings" DESC, "country" ASC;
